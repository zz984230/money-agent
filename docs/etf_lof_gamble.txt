class LOFETFAnalyzer:
    def __init__(self):

        self.commodity_lof_list = []  # 大宗商品LOF
        self.etf_list = []  # ETF列表
        
    def get_all_lof_etf_codes(self):
        """获取所有LOF和ETF代码"""
        try:
            # 获取LOF基金列表
            fund_lof_df = ak.fund_em_fund_name()
            lof_funds = fund_lof_df[fund_lof_df['基金类型'].str.contains('LOF')]
            
            # 获取ETF列表
            etf_df = ak.fund_etf_category_sina(symbol="ETF基金")
            
            # 筛选大宗商品相关
            commodity_keywords = ['商品', '黄金', '原油', '石油', '白银', '有色金属',
                                '能源', '农产品', '豆粕', '煤炭', '钢铁']
            
            self.commodity_lof_list = []
            for _, row in lof_funds.iterrows():
                name = row['基金简称']
                if any(keyword in name for keyword in commodity_keywords):
                    self.commodity_lof_list.append({
                        'code': row['基金代码'],
                        'name': name,
                        'type': 'LOF'
                    })
            
            print(f"找到{len(self.commodity_lof_list)}个大宗商品LOF")
            return self.commodity_lof_list
            
        except Exception as e:
            print(f"获取基金列表失败: {e}")
            return []
    
    def get_historical_data(self, symbol, start_date=None, end_date=None):
        """获取历史行情数据"""
        if start_date is None:
            start_date = (datetime.now() - timedelta(days=365)).strftime('%Y%m%d')
        if end_date is None:
            end_date = datetime.now().strftime('%Y%m%d')
        
        try:
            # 尝试多种数据源
            # 方法1: 基金历史数据
            df = ak.fund_em_etf_hist(symbol=symbol, period="daily", 
                                     start_date=start_date, end_date=end_date)
            if len(df) > 0:
                df = df[['日期', '开盘', '收盘', '最高', '最低', '成交量', '成交额']]
                df.columns = ['date', 'open', 'close', 'high', 'low', 'volume', 'amount']
                df['date'] = pd.to_datetime(df['date'])
                df.set_index('date', inplace=True)
                return df
            
            # 方法2: 备用数据源
            df = ak.fund_etf_hist_sina(symbol=symbol)
            if len(df) > 0:
                df.index = pd.to_datetime(df.index)
                return df
                
        except Exception as e:
            print(f"获取{symbol}数据失败: {e}")
            
        return None

class VolatilityDetector:
    def __init__(self):
        self.volatility_events = []
    
    def detect_sudden_moves(self, price_series, window=3, threshold=0.15):
        """
        检测突增突降
        window: 观察窗口（天）
        threshold: 阈值（15%）
        """
        returns = price_series.pct_change(window)
        abnormal_dates = []
        abnormal_info = []
        
        for i in range(window, len(returns)):
            if abs(returns.iloc[i]) > threshold:
                date = returns.index[i]
                start_price = price_series.iloc[i-window]
                end_price = price_series.iloc[i]
                change_pct = returns.iloc[i]
                
                # 计算波动特征
                volatility = price_series.iloc[i-window:i+1].std() / price_series.iloc[i-window:i+1].mean()
                
                abnormal_dates.append(date)
                abnormal_info.append({
                    'date': date,
                    'return': change_pct,
                    'start_price': start_price,
                    'end_price': end_price,
                    'volatility': volatility,
                    'window': window
                })
        
        return abnormal_dates, abnormal_info
    
    def multi_timeframe_analysis(self, price_series):
        """多时间框架分析"""
        results = {}
        
        # 不同时间窗口检测
        for window in [2, 3, 5]:
            dates, info = self.detect_sudden_moves(price_series, window=window, threshold=0.1)
            results[f'window_{window}'] = {
                'dates': dates,
                'info': info,
                'count': len(dates)
            }
        
        # 计算综合波动指标
        returns = price_series.pct_change()
        results['metrics'] = {
            'max_1d_return': returns.max(),
            'min_1d_return': returns.min(),
            'volatility_20d': returns.rolling(20).std().mean(),
            'sharp_ratio': returns.mean() / returns.std() * np.sqrt(252) if returns.std() > 0 else 0
        }
        
        return results

class PredictiveFactorAnalyzer:
    def __init__(self):
        self.factors = {}
        
    def calculate_technical_factors(self, df):
        """计算技术因子"""
        factors = pd.DataFrame(index=df.index)
        
        # 价格动量因子
        factors['momentum_5'] = df['close'].pct_change(5)
        factors['momentum_10'] = df['close'].pct_change(10)
        factors['momentum_20'] = df['close'].pct_change(20)
        
        # 波动率因子
        factors['volatility_20'] = df['close'].pct_change().rolling(20).std()
        factors['atr_14'] = talib.ATR(df['high'], df['low'], df['close'], timeperiod=14)
        
        # 成交量因子
        factors['volume_ratio'] = df['volume'] / df['volume'].rolling(20).mean()
        factors['volume_ma_5'] = df['volume'].rolling(5).mean() / df['volume'].rolling(20).mean()
        
        # 技术指标
        factors['rsi_14'] = talib.RSI(df['close'], timeperiod=14)
        factors['macd'] = talib.MACD(df['close'])[0]
        factors['bollinger_bandwidth'] = (
            talib.BBANDS(df['close'])[0] - talib.BBANDS(df['close'])[2]
        ) / talib.BBANDS(df['close'])[1]
        
        # 价量背离
        price_change = df['close'].pct_change(5)
        volume_change = df['volume'].pct_change(5)
        factors['pv_divergence'] = price_change - volume_change
        
        return factors
    
    def calculate_liquidity_factors(self, df):
        """计算流动性因子"""
        factors = pd.DataFrame(index=df.index)
        
        # 换手率（估算）
        if 'amount' in df.columns:
            factors['turnover'] = df['amount'] / df['close'] / 1e6  # 简化换手率
        
        # 价差指标
        factors['spread_pct'] = (df['high'] - df['low']) / df['close']
        
        # 流动性冲击
        daily_return = df['close'].pct_change()
        factors['liquidity_impact'] = daily_return.abs() / (df['volume'] + 1e-6)
        
        return factors
    
    def calculate_commodity_specific_factors(self, df):
        """大宗商品特有因子"""
        factors = pd.DataFrame(index=df.index)
        
        # 期限结构相关（简化版）
        factors['price_trend'] = df['close'].rolling(20).apply(
            lambda x: 1 if (x[-1] > x[0]) else -1
        )
        
        # 波动聚集性
        returns = df['close'].pct_change()
        factors['vol_clustering'] = returns.rolling(5).std() / returns.rolling(20).std()
        
        return factors
    
    def build_prediction_model(self, factors, target_events, lookback_days=10):
        """
        构建预测模型
        target_events: 突增突降事件日期列表
        lookback_days: 事件前观察天数
        """
        # 创建标签
        labels = pd.Series(0, index=factors.index)
        for event_date in target_events:
            if event_date in factors.index:
                # 标记事件前N天为正样本
                pred_date = event_date - pd.Timedelta(days=1)
                if pred_date in factors.index:
                    labels.loc[pred_date] = 1
        
        # 对齐数据
        aligned_data = pd.concat([factors, labels], axis=1)
        aligned_data.columns = list(factors.columns) + ['label']
        aligned_data = aligned_data.dropna()
        
        if len(aligned_data) == 0:
            return None, None
        
        # 特征重要性分析
        from sklearn.ensemble import RandomForestClassifier
        from sklearn.model_selection import train_test_split
        
        X = aligned_data.iloc[:, :-1]
        y = aligned_data['label']
        
        # 处理类别不平衡
        if y.sum() > 0:
            X_train, X_test, y_train, y_test = train_test_split(
                X, y, test_size=0.3, random_state=42, stratify=y
            )
            
            clf = RandomForestClassifier(n_estimators=100, random_state=42)
            clf.fit(X_train, y_train)
            
            # 特征重要性
            feature_importance = pd.DataFrame({
                'feature': X.columns,
                'importance': clf.feature_importances_
            }).sort_values('importance', ascending=False)
            
            return clf, feature_importance
        
        return None, None

class LOFETFTradingSystem:
    def __init__(self):
        self.data_fetcher = LOFETFAnalyzer()
        self.detector = VolatilityDetector()
        self.factor_analyzer = PredictiveFactorAnalyzer()
        self.results = {}
    
    def run_analysis(self, test_count=10):
        """运行完整分析"""
        # 1. 获取大宗商品LOF列表
        print("步骤1: 获取大宗商品LOF列表...")
        lof_list = self.data_fetcher.get_all_lof_etf_codes()
        
        # 限制测试数量
        lof_list = lof_list[:test_count]
        
        # 2. 对每个LOF进行分析
        for lof in lof_list:
            print(f"\n分析 {lof['name']} ({lof['code']})...")
            
            # 获取历史数据
            df = self.data_fetcher.get_historical_data(lof['code'])
            if df is None or len(df) < 100:
                print(f"  数据不足，跳过")
                continue
            
            # 检测突增突降
            abnormal_dates, abnormal_info = self.detector.detect_sudden_moves(
                df['close'], window=3, threshold=0.15
            )
            
            if len(abnormal_dates) > 0:
                print(f"  发现 {len(abnormal_dates)} 个异常波动事件")
                
                # 计算预测因子
                tech_factors = self.factor_analyzer.calculate_technical_factors(df)
                liq_factors = self.factor_analyzer.calculate_liquidity_factors(df)
                commodity_factors = self.factor_analyzer.calculate_commodity_specific_factors(df)
                
                # 合并因子
                all_factors = pd.concat([tech_factors, liq_factors, commodity_factors], axis=1)
                
                # 构建预测模型
                model, feature_importance = self.factor_analyzer.build_prediction_model(
                    all_factors, abnormal_dates
                )
                
                if model is not None:
                    # 保存结果
                    self.results[lof['code']] = {
                        'name': lof['name'],
                        'abnormal_events': len(abnormal_dates),
                        'events_info': abnormal_info,
                        'factors': all_factors,
                        'feature_importance': feature_importance,
                        'model': model
                    }
                    
                    print(f"  重要因子: {feature_importance.head(5)['feature'].tolist()}")
    
    def generate_signals(self, code, current_factors):
        """生成交易信号"""
        if code not in self.results:
            return None
        
        model = self.results[code]['model']
        prediction = model.predict_proba(current_factors.reshape(1, -1))
        
        return {
            'probability': prediction[0][1],  # 上涨概率
            'signal': 1 if prediction[0][1] > 0.7 else -1 if prediction[0][1] < 0.3 else 0
        }
    
    def get_top_factors_across_funds(self):
        """获取所有基金中的重要因子"""
        all_importances = []
        for code, result in self.results.items():
            if 'feature_importance' in result:
                importance_df = result['feature_importance'].copy()
                importance_df['fund'] = code
                importance_df['fund_name'] = result['name']
                all_importances.append(importance_df)
        
        if not all_importances:
            return pd.DataFrame()
        
        combined = pd.concat(all_importances)
        
        # 计算因子在所有基金中的平均重要性
        factor_ranking = combined.groupby('feature')['importance'].agg(['mean', 'count']).reset_index()
        factor_ranking = factor_ranking.sort_values('mean', ascending=False)
        
        return factor_ranking

根据大宗商品LOF/ETF的特性，建议重点关注以下因子组合：

技术面因子：
成交量异常放大 - 量比 > 2，且持续3天

价格波动率压缩 - ATR达到近期低点后开始扩张

RSI背离 - 价格创新低但RSI未创新低（底部）/价格创新高但RSI未创新高（顶部）

流动性因子：
买卖价差突然收窄 - 预示大资金进入

订单簿失衡 - 买一卖一量比异常

期现价差变化 - 可通过相关期货数据获取
